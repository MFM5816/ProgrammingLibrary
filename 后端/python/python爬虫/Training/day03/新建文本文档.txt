create database marketdb charset utf8;
use marketdb;
create table product(
id int primary key,
name varchar(30),
price decimal(20,2),
comment varchar(50),
time date
)charset=utf8;

create table type(
id int primary key,
type varchar(30),
comment varchar(50)
)charset=utf8;

create table middle(
pid int,
tid int,
foreign key(pid) references product(id) on delete cascade on update cascade,
foreign key(tid) references type(id) on delete cascade on update cascade
)charset=utf8;
